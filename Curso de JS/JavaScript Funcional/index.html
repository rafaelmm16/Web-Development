index.js - Arrow Functions, IIFE e Spread

// 1. IIFE (Immediately Invoked Function Expression)
(function (x, y){
    console.log("IIFE simples:", x + y);
})(5, 5);

// 2. Function Expression
const soma = function(x, y){
    return x + y;
};
console.log("Function expression:", soma(2, 5));

// 3. Arrow Function
const somav2 = (x, y) => x + y;
console.log("Arrow function 1 linha:", somav2(2, 5));

// 4. Spread Operator
const somav5 = (...numeros) => {
    let aux = 0;
    for(let i of numeros){
        aux += i;
    }
    return aux;
};
console.log("SPREAD (2, 2, 10, 30, 55):", somav5(2, 2, 10, 30, 55));

desafio.js - Funções Aritméticas em 1 Linha

// Funções em uma única linha com Arrow Functions
const soma = (a, b) => a + b;
const subtracao = (a, b) => a - b;
const multiplicacao = (a, b) => a * b;
const divisao = (a, b) => a / b;

const a = 10;
const b = 5;

console.log("Soma (10 + 5):", soma(a, b));
console.log("Subtração (10 - 5):", subtracao(a, b));
console.log("Multiplicação (10 * 5):", multiplicacao(a, b));
console.log("Divisão (10 / 5):", divisao(a, b));

map.js - Mapeamento de Arrays e Objetos

function mapNotas(notas, funcao) {
  return notas.map((nota) => funcao(nota));
}

// Array de números
const notas = [10, 9, 8, 7, 6];
const dobradas = mapNotas(notas, (nota) => nota * 2);
console.log("Notas Dobradas:", dobradas);

// Array de objetos
const boletim = [
  { nota: 10, materia: "Matemática" },
  { nota: 9, materia: "História" },
  { nota: 8, materia: "Português" },
  { nota: 7, materia: "Ciências" },
  { nota: 6, materia: "Inglês" }
];

const boletimDourado = mapNotas(boletim, (item) => ({
  nota: item.nota * 2,
  materia: item.materia
}));
console.log("Boletim Mapeado:", JSON.stringify(boletimDourado, null, 2));
